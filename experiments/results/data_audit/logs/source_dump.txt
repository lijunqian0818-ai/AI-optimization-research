def stock_zh_a_hist_em(

    adjust: str = "",
) -> pd.DataFrame:
    """
    东方财富网-行情首页-沪深京 A 股-每日行情
    https://quote.eastmoney.com/concept/sh603777.html?from=classic
    :param symbol: 股票代码
    :type symbol: str
    :param period: choice of {'daily', 'weekly', 'monthly'}
    :type period: str
    :param start_date: 开始日期
    :type start_date: str
    :param end_date: 结束日期
    :type end_date: str
    :param adjust: choice of {"qfq": "前复权", "hfq": "后复权", "": "不复权"}
    :type adjust: str
    :return: 每日行情
    :rtype: pandas.DataFrame

    adjust_dict = {"qfq": "1", "hfq": "2", "": "0"}

    temp_df["成交额"] = pd.to_numeric(temp_df["成交额"], errors="coerce")

### stock_zh_a_daily_sina(symbol: str = 'sh603843', start_date: str = '19900101', end_date: str = '21000118', adjust: str = '') -> pandas.DataFrame
file: C:\Users\lijun\Desktop\AI-Optimization-Research\experiments\.venv\Lib\site-packages\akshare\stock\quote\stock_zh_a_sina.py
urls: ['https://finance.sina.com.cn/realstock/company/sh603843/nc.shtml']
def stock_zh_a_daily_sina(
    symbol: str = "sh603843",
    start_date: str = "19900101",
    end_date: str = "21000118",
    adjust: str = "",
) -> pd.DataFrame:
    """
    新浪财经-A 股-个股的历史行情数据，大量抓取容易封 IP
    https://finance.sina.com.cn/realstock/company/sh603843/nc.shtml
    :param symbol: sh600000
    :type symbol: str
    :param start_date: 20201103；开始日期
    :type start_date: str
    :param end_date: 20201103；结束日期
    :type end_date: str
    :param adjust: 默认为空：返回不复权的数据；qfq: 返回前复权后的数据；hfq: 返回后复权后的数据；hfq-factor: 返回后复权因子；qfq-factor: 返回前复权因子
    :type adjust: str
    :return: 行情数据
    :rtype: pandas.DataFrame
    """

    def _fq_factor(method: str) -> pd.DataFrame:
        if method == "hfq":
            r = http.get(zh_sina_a_stock_hfq_url.format(symbol))
            hfq_factor_df = pd.DataFrame(
                ast.literal_eval(r.text.split("=")[1].split("\n")[0])["data"]
            )
            if hfq_factor_df.shape[0] == 0:
                raise ValueError("sina hfq factor not available")
            hfq_factor_df.columns = ["date", "hfq_factor"]
            hfq_factor_df.index = pd.to_datetime(hfq_factor_df.date)
            del hfq_factor_df["date"]
            hfq_factor_df = hfq_factor_df.reset_index()
            return hfq_factor_df
        else:
            r = http.get(zh_sina_a_stock_qfq_url.format(symbol))
            qfq_factor_df = pd.DataFrame(
                ast.literal_eval(r.text.split("=")[1].split("\n")[0])["data"]
            )
            if qfq_factor_df.shape[0] == 0:
                raise ValueError("sina hfq factor not available")
            qfq_factor_df.columns = ["date", "qfq_factor"]
            qfq_factor_df.index = pd.to_datetime(qfq_factor_df.date)
            del qfq_factor_df["date"]
            qfq_factor_df = qfq_factor_df.reset_index()
            return qfq_factor_df

    if adjust in ("hfq-factor", "qfq-factor"):
        return _fq_factor(adjust.split("-")[0])

    r = http.get(zh_sina_a_stock_hist_url.format(symbol))
    if r.status_code == 404:
        raise InvalidParameterError(
            f"新浪财经没有 {symbol} 的历史行情，请检查股票代码，格式如 sh600000、sz000001、bj920000"
        )
    js_code = py_mini_racer.MiniRacer()
    js_code.eval(hk_js_decode)
    dict_list = js_code.call(
        "d", r.text.split("=")[1].split(";")[0].replace('"', "")
    )  # 执行js解密代码
    data_df = pd.DataFrame(dict_list)
    # 与股本、复权因子表一样用不带时区的 Timestamp 作索引；混用 datetime.date 时 pandas 合并会
    # 发出 Pandas4Warning，pandas 4 起将报错。上游日期形如 1999-11-10T00:00:00.000Z（UTC 零点），
    # 去掉时区后日期不变。返回前 reset_index 后再转为 date，结果类型不变
    data_df.index = pd.to_datetime(data_df["date"], errors="coerce").dt.tz_localize(
        None
    )
    del data_df["date"]
    try:
        del data_df["prevclose"]
    except Exception:
        pass
    try:
        del data_df["postVol"]
        del data_df["postAmt"]
    except Exception:
        pass
    data_df = data_df.astype("float")
    # 同一日期只保留一行；不能按开高低收量去重，冷门股不同交易日的价量常常完全相同（#7477）
    data_df = data_df[~data_df.index.duplicated(keep="first")].sort_index()
    r = http.get(zh_sina_a_stock_amount_url.format(symbol, symbol))
    # 已退市的股票有历史行情，但股本接口返回 (null)；此时照常返回行情，股本与换手率留空。
    # 原先直接解析空文本，报出与原因无关的 JSONDecodeError
    if "[" in r.text:
        amount_data_json = jsonish.loads(
            r.text[r.text.find("[") : r.text.rfind("]") + 1]
        )
        price_columns = None
    else:
        warnings.warn(
            f"新浪财经没有 {symbol} 的股本数据（通常是已退市的股票），"
            "outstanding_share 与 turnover 为空"
        )
        amount_data_json = []
        price_columns = ["open", "high", "low", "close", "volume", "amount"]
    # 上游每条记录为 {"date": ..., "amount": ...}，amount 为股本（万股）
    amount_data_df = pd.DataFrame(amount_data_json, columns=["date", "amount"]).rename(
        columns={"amount": "outstanding_share"}
    )
    amount_data_df.index = pd.to_datetime(amount_data_df.date)
    del amount_data_df["date"]
    # 股本按变动日期记录，对齐到交易日；不再 outer merge，以免带进非交易日的行
    temp_df = data_df.join(ffill_to_index(amount_data_df, data_df.index))
    temp_df = temp_df.astype(float)
    temp_df["outstanding_share"] = temp_df["outstanding_share"] * 10000
    temp_df["turnover"] = temp_df["volume"] / temp_df["outstanding_share"]
    temp_df.columns = [
        "open",
        "high",
        "low",
        "close",
        "volume",
        "amount",
        "outstanding_share",
        "turnover",
    ]
    if adjust == "":
        temp_df = temp_df[start_date:end_date]
        temp_df["open"] = round(temp_df["open"], 2)
        temp_df["high"] = round(temp_df["high"], 2)
        temp_df["low"] = round(temp_df["low"], 2)
        temp_df["close"] = round(temp_df["close"], 2)
        temp_df = temp_df.dropna(subset=price_columns)
        temp_df = temp_df.reset_index()
        temp_df["date"] = pd.to_datetime(temp_df["date"], errors="coerce").dt.date
        return temp_df
    if adjust == "hfq":
        res = http.get(zh_sina_a_stock_hfq_url.format(symbol))
        hfq_factor_df = pd.DataFrame(
            ast.literal_eval(res.text.split("=")[1].split("\n")[0])["data"]
        )
        hfq_factor_df.columns = ["date", "hfq_factor"]
        hfq_factor_df.index = pd.to_datetime(hfq_factor_df.date)
        del hfq_factor_df["date"]
        temp_df = temp_df.join(ffill_to_index(hfq_factor_df, temp_df.index))
        temp_df = temp_df.astype(float)
        temp_df = temp_df.dropna(
            subset=None if price_columns is None else price_columns + ["hfq_factor"]
        )
        temp_df["open"] = temp_df["open"] * temp_df["hfq_factor"]
        temp_df["high"] = temp_df["high"] * temp_df["hfq_factor"]
        temp_df["close"] = temp_df["close"] * temp_df["hfq_factor"]
        temp_df["low"] = temp_df["low"] * temp_df["hfq_factor"]
        temp_df = temp_df.iloc[:, :-1]
        temp_df = temp_df[start_date:end_date]
        temp_df["open"] = round(temp_df["open"], 2)
        temp_df["high"] = round(temp_df["high"], 2)
        temp_df["low"] = round(temp_df["low"], 2)
        temp_df["close"] = round(temp_df["close"], 2)
        temp_df = temp_df.dropna(subset=price_columns)
        temp_df = temp_df.reset_index()
        temp_df["date"] = pd.to_datetime(temp_df["date"], errors="coerce").dt.date
        return temp_df

    if adjust == "qfq":
        res = http.get(zh_sina_a_stock_qfq_url.format(symbol))
        qfq_factor_df = pd.DataFrame(
            ast.literal_eval(res.text.split("=")[1].split("\n")[0])["data"]
        )
        qfq_factor_df.columns = ["date", "qfq_factor"]
        qfq_factor_df.index = pd.to_datetime(qfq_factor_df.date)
        del qfq_factor_df["date"]
        temp_df = temp_df.join(ffill_to_index(qfq_factor_df, temp_df.index))
        temp_df = temp_df.astype(float)
        temp_df = temp_df.dropna(
            subset=None if price_columns is None else price_columns + ["qfq_factor"]
        )
        temp_df["open"] = temp_df["open"] / temp_df["qfq_factor"]
        temp_df["high"] = temp_df["high"] / temp_df["qfq_factor"]
        temp_df["close"] = temp_df["close"] / temp_df["qfq_factor"]
        temp_df["low"] = temp_df["low"] / temp_df["qfq_factor"]
        temp_df = temp_df.iloc[:, :-1]
        temp_df = temp_df[start_date:end_date]
        temp_df["open"] = round(temp_df["open"], 2)
        temp_df["high"] = round(temp_df["high"], 2)
        temp_df["low"] = round(temp_df["low"], 2)
        temp_df["close"] = round(temp_df["close"], 2)
        temp_df = temp_df.dropna(subset=price_columns)
        temp_df = temp_df.reset_index()
        temp_df["date"] = pd.to_datetime(temp_df["date"], errors="coerce").dt.date
        return temp_df
    return pd.DataFrame()


### stock_zh_a_hist_tx(symbol: str = 'sz000001', start_date: str = '19000101', end_date: str = '20500101', adjust: str = '') -> pandas.DataFrame

def stock_zh_a_hist_tx(
    symbol: str = "sz000001",
    start_date: str = "19000101",
    end_date: str = "20500101",
    adjust: str = "",
) -> pd.DataFrame:
    """
    腾讯证券-日频-股票历史数据
    https://gu.qq.com/sh000919/zs
    :param symbol: 带市场标识的股票或者指数代码
    :type symbol: str
    :param start_date: 开始日期，格式 YYYYMMDD 或 YYYY-MM-DD
    :type start_date: str
    :param end_date: 结束日期，格式 YYYYMMDD 或 YYYY-MM-DD
    :type end_date: str
    :param adjust: choice of {"qfq": "前复权", "hfq": "后复权", "": "不复权"}
    :type adjust: str
    :return: 历史行情数据，其中 volume 统一为股，amount 统一为元
    :rtype: pandas.DataFrame
    """
    symbol = _normalize_tx_symbol(symbol=symbol)
    start = _parse_tx_date(value=start_date, name="start_date")
    end = _parse_tx_date(value=end_date, name="end_date")
    if start > end:
        raise InvalidParameterError(
            f"start_date（{start_date}）不能晚于 end_date（{end_date}）"
        )
    init_start = _parse_tx_date(
        value=get_tx_start_year(symbol=symbol), name="get_tx_start_year"
    )
    start = max(start, init_start)
    url = "https://proxy.finance.qq.com/ifzqgtimg/appstock/app/newfqkline/get"
    # 每次请求覆盖两个自然年（约 490 个交易日，低于单次 640 条的上限），按两年步进
    range_end = min(end.year, datetime.date.today().year) + 1
    big_df = pd.DataFrame()
    tqdm = get_tqdm()
    for year in tqdm(range(start.year, range_end, _TX_YEARS_PER_REQUEST), leave=False):
        params = {
            "_var": f"kline_day{adjust}{year}",
            "param": f"{symbol},day,{year}-01-01,{year + 1}-12-31,640,{adjust}",
            "r": "0.8205512681390605",
        }
        r = http.get(url, params=params)
        data_text = r.text
        data_json = jsonish.loads(data_text[data_text.find("={") + 1 :])["data"][symbol]
        if "day" in data_json.keys():
            temp_df = pd.DataFrame(data_json["day"])
        elif "hfqday" in data_json.keys():
            temp_df = pd.DataFrame(data_json["hfqday"])
        else:
            temp_df = pd.DataFrame(data_json["qfqday"])
        big_df = pd.concat([big_df, temp_df], ignore_index=True)
    if big_df.empty:
        return pd.DataFrame(columns=_TX_COLUMNS)
    big_df = big_df.iloc[:, _TX_KLINE_FIELDS]
    big_df.columns = _TX_COLUMNS
    big_df["date"] = pd.to_datetime(big_df["date"], errors="coerce").dt.date
    big_df["open"] = pd.to_numeric(big_df["open"], errors="coerce")
    big_df["close"] = pd.to_numeric(big_df["close"], errors="coerce")
    big_df["high"] = pd.to_numeric(big_df["high"], errors="coerce")
    big_df["low"] = pd.to_numeric(big_df["low"], errors="coerce")
    big_df["volume"] = pd.to_numeric(big_df["volume"], errors="coerce")
    big_df["turnover"] = pd.to_numeric(big_df["turnover"], errors="coerce")
    big_df["amount"] = pd.to_numeric(big_df["amount"], errors="coerce")
    big_df["volume"] = big_df["volume"] * _tx_volume_multiplier(symbol)
    big_df["turnover"] = big_df["turnover"] / 100
    big_df["amount"] = big_df["amount"] * 10000
    big_df = big_df.drop_duplicates(ignore_index=True)
    big_df.index = pd.to_datetime(big_df["date"], errors="coerce")
    big_df = big_df.sort_index()
    big_df = big_df[pd.Timestamp(start) : pd.Timestamp(end)]
    big_df = big_df.reset_index(drop=True)
    return big_df

def index_stock_cons(symbol: str = "399639") -> pd.DataFrame:
    """
    最新股票指数的成份股目录
    https://vip.stock.finance.sina.com.cn/corp/view/vII_NewestComponent.php?page=1&indexid=399639
    :param symbol: 指数代码，可以通过 ak.index_stock_info() 函数获取
    :type symbol: str
    :return: 最新股票指数的成份股目录
    :rtype: pandas.DataFrame
    """
    url = f"https://vip.stock.finance.sina.com.cn/corp/go.php/vII_NewestComponent/indexid/{symbol}.phtml"
    r = http.get(url)
    r.encoding = "gb2312"
    soup = BeautifulSoup(r.text, "lxml")
    page_num = (
        soup.find(attrs={"class": "table2"})
        .find("td")
        .find_all("a")[-1]["href"]
        .split("page=")[-1]
        .split("&")[0]
    )
    if page_num == "#":
        temp_df = pd.read_html(StringIO(r.text), header=0, skiprows=1)[3].iloc[:, :3]
        temp_df["品种代码"] = temp_df["品种代码"].astype(str).str.zfill(6)
        return temp_df

    temp_df = pd.DataFrame()
    for page in range(1, int(page_num) + 1):
        url = f"https://vip.stock.finance.sina.com.cn/corp/view/vII_NewestComponent.php?page={page}&indexid={symbol}"
        r = http.get(url)
        r.encoding = "gb2312"
        temp_df = pd.concat(
            objs=[temp_df, pd.read_html(StringIO(r.text), header=1)[3]],
            ignore_index=True,
        )
    temp_df = temp_df.iloc[:, :3]
    temp_df["品种代码"] = temp_df["品种代码"].astype(str).str.zfill(6)
    return temp_df
